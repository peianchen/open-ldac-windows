cmake_minimum_required(VERSION 3.20)
project(windows_ldac VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(ldac_encoder STATIC vendor/libldac/src/ldaclib.c vendor/libldac/src/ldacBT.c)
target_include_directories(ldac_encoder PUBLIC vendor/libldac/inc)
if(MSVC)
  # Upstream uses one GNU unused attribute and includes unistd.h without POSIX calls.
  # Keep the pinned upstream source unchanged; scope compatibility to this target.
  target_include_directories(ldac_encoder PRIVATE compat/msvc)
  target_compile_options(ldac_encoder PRIVATE /FI${CMAKE_CURRENT_SOURCE_DIR}/compat/msvc/compiler.h)
endif()
if(NOT MSVC)
  target_link_libraries(ldac_encoder PUBLIC m)
endif()
add_library(wldac STATIC src/wldac.c src/avdtp.c src/avdtp_session.c src/avdtp_reassembly.c)
target_include_directories(wldac PUBLIC include)
target_link_libraries(wldac PUBLIC ldac_encoder)
add_executable(wldac_cli tools/cli.c)
target_link_libraries(wldac_cli PRIVATE wldac)
enable_testing()
add_executable(core_tests tests/core_tests.c)
target_link_libraries(core_tests PRIVATE wldac)
add_test(NAME core_tests COMMAND core_tests)
add_executable(avdtp_session_tests tests/avdtp_session_tests.c)
target_link_libraries(avdtp_session_tests PRIVATE wldac)
add_test(NAME avdtp_session_tests COMMAND avdtp_session_tests)
add_executable(avdtp_reassembly_tests tests/avdtp_reassembly_tests.c)
target_link_libraries(avdtp_reassembly_tests PRIVATE wldac)
add_test(NAME avdtp_reassembly_tests COMMAND avdtp_reassembly_tests)
add_library(wl_playback STATIC src/playback.c src/wav_source.c src/reconnect.c)
target_include_directories(wl_playback PUBLIC include)
target_link_libraries(wl_playback PUBLIC wldac)
# Fault-injected coordinator test supplies its own playback entry point.
add_executable(reconnect_tests tests/reconnect_tests.c src/reconnect.c)
target_include_directories(reconnect_tests PRIVATE include vendor/libldac/inc)
add_test(NAME reconnect_tests COMMAND reconnect_tests)
add_executable(playback_tests tests/playback_tests.c)
target_link_libraries(playback_tests PRIVATE wl_playback)
add_test(NAME playback_tests COMMAND playback_tests)
add_executable(wav_source_tests tests/wav_source_tests.c)
target_link_libraries(wav_source_tests PRIVATE wl_playback)
add_test(NAME wav_source_tests COMMAND wav_source_tests)
if(WIN32)
  add_executable(wldac_probe tools/native_probe.c)
  target_link_libraries(wldac_probe PRIVATE wldac setupapi)
  add_library(wl_native_io STATIC src/win_native_io.c)
  target_compile_definitions(wl_native_io PRIVATE _WIN32_WINNT=0x0a00)
  target_link_libraries(wl_native_io PUBLIC wl_playback setupapi)
  add_executable(wldac_player tools/player.c)
  target_link_libraries(wldac_player PRIVATE wl_native_io advapi32 shell32 avrt)
  # The test includes the production adapter to reach its internal OS-I/O seam.
  add_executable(native_io_tests tests/windows/native_io_tests.c)
  target_compile_definitions(native_io_tests PRIVATE _WIN32_WINNT=0x0a00)
  target_link_libraries(native_io_tests PRIVATE wl_playback setupapi)
  add_test(NAME native_io_tests COMMAND native_io_tests)
  set_tests_properties(native_io_tests PROPERTIES TIMEOUT 60)
endif()

# Keep lifecycle sanitizers independent of the pinned third-party encoder.
add_library(ol_lifetime STATIC src/lifetime.c)
target_include_directories(ol_lifetime PUBLIC include)
add_executable(lifetime_tests tests/lifetime_tests.c)
target_link_libraries(lifetime_tests PRIVATE ol_lifetime)
add_test(NAME lifetime_tests COMMAND lifetime_tests)
add_executable(native_adapter_tests tests/native/native_adapter_tests.c
  driver/transport/connection.c driver/transport/client.c
  driver/transport/control.c driver/transport/queue.c)
target_compile_definitions(native_adapter_tests PRIVATE OL_NATIVE_HOST_TEST=1)
target_include_directories(native_adapter_tests PRIVATE driver/transport tests/native)
target_link_libraries(native_adapter_tests PRIVATE ol_lifetime)
add_test(NAME native_adapter_tests COMMAND native_adapter_tests)
option(OL_LIFETIME_SANITIZERS "ASan/UBSan for lifecycle, native and protocol host tests" OFF)
if(OL_LIFETIME_SANITIZERS AND NOT MSVC)
  foreach(target ol_lifetime lifetime_tests native_adapter_tests wldac avdtp_session_tests avdtp_reassembly_tests wl_playback playback_tests wav_source_tests reconnect_tests)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -g)
  endforeach()
  target_link_options(reconnect_tests PRIVATE -fsanitize=address,undefined)
  target_link_options(lifetime_tests PRIVATE -fsanitize=address,undefined)
  target_link_options(native_adapter_tests PRIVATE -fsanitize=address,undefined)
  # wldac is also linked by the CLI/core tests; propagate the runtime dependency.
  # The pinned third-party encoder is not instrumented by this option.
  target_link_options(wldac PUBLIC -fsanitize=address,undefined)
endif()
