cmake_minimum_required(VERSION 3.20)
project(windows_ldac VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(ldac_encoder STATIC vendor/libldac/src/ldaclib.c vendor/libldac/src/ldacBT.c)
target_include_directories(ldac_encoder PUBLIC vendor/libldac/inc)
if(MSVC)
  # Upstream uses one GNU unused attribute and includes unistd.h without POSIX calls.
  # Keep the pinned upstream source unchanged; scope compatibility to this target.
  target_include_directories(ldac_encoder PRIVATE compat/msvc)
  target_compile_options(ldac_encoder PRIVATE /FI${CMAKE_CURRENT_SOURCE_DIR}/compat/msvc/compiler.h)
endif()
if(NOT MSVC)
  target_link_libraries(ldac_encoder PUBLIC m)
endif()
add_library(wldac STATIC src/wldac.c src/avdtp.c)
target_include_directories(wldac PUBLIC include)
target_link_libraries(wldac PUBLIC ldac_encoder)
add_executable(wldac_cli tools/cli.c)
target_link_libraries(wldac_cli PRIVATE wldac)
enable_testing()
add_executable(core_tests tests/core_tests.c)
target_link_libraries(core_tests PRIVATE wldac)
add_test(NAME core_tests COMMAND core_tests)
if(WIN32)
  add_executable(wldac_probe tools/native_probe.c)
  target_link_libraries(wldac_probe PRIVATE wldac setupapi)
endif()
